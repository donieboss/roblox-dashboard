const ROBLOX_GAMES = "https://games.roblox.com"; const ROBLOX_USERS = "https://users.roblox.com"; const ROBLOX_INVENTORY = "https://inventory.roblox.com"; export default { async fetch(request) { const url = new URL(request.url); if (request.method === "OPTIONS") { return new Response(null, { headers: corsHeaders() }); } try { // ------------------------- // HEALTH CHECK // ------------------------- if (url.pathname === "/") { return json({ success: true, message: "DONIE Worker is online" }); } // ------------------------- // USERNAME -> USER ID // ------------------------- if (url.pathname === "/user") { const username = url.searchParams.get("username"); if (!username) { return json({ success: false, error: "Missing username" }, 400); } const response = await fetch( `${ROBLOX_USERS}/v1/usernames/users`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ usernames: [username], excludeBannedUsers: false }) } ); if (!response.ok) { throw new Error( `Roblox returned HTTP ${response.status}` ); } const data = await response.json(); const user = data.data?.[0]; if (!user) { return json({ success: false, error: "Roblox user not found" }, 404); } return json({ success: true, user: { id: user.id, name: user.name, displayName: user.displayName } }); } // ------------------------- // PROFILE // ------------------------- if (url.pathname === "/profile") { const userId = url.searchParams.get("userId"); if (!userId) { return json({ success: false, error: "Missing userId" }, 400); } const response = await fetch( `${ROBLOX_USERS}/v1/users/${encodeURIComponent(userId)}` ); if (!response.ok) { return json({ success: false, error: "User not found" }, 404); } const user = await response.json(); return json({ success: true, user: { id: user.id, name: user.name, displayName: user.displayName, description: user.description || "", created: user.created } }); } // ------------------------- // COLLECTIBLES + RAP // ------------------------- if (url.pathname === "/value") { const userId = url.searchParams.get("userId"); if (!userId) { return json({ success: false, error: "Missing userId" }, 400); } const result = await getCollectibles(userId); return json({ success: true, userId: Number(userId), totalRAP: result.totalRAP, limitedCount: result.items.length, items: result.items, warnings: result.warnings }); } // ------------------------- // SERVERS // ------------------------- if (url.pathname === "/servers") { const placeId = url.searchParams.get("placeId") || "920587237"; const limit = Math.min( Number(url.searchParams.get("limit") || 100), 100 ); const apiUrl = `${ROBLOX_GAMES}/v1/games/${encodeURIComponent(placeId)}` + `/servers/Public?sortOrder=Desc&limit=${limit}`; const response = await fetch(apiUrl); if (!response.ok) { throw new Error( `Roblox server API returned ${response.status}` ); } const data = await response.json(); const servers = (data.data || []).map(server => ({ id: server.id, playing: Number(server.playing || 0), maxPlayers: Number(server.maxPlayers || 0), ping: server.ping ?? null, fps: server.fps ?? null })); return json({ success: true, count: servers.length, data: servers }); } return json({ success: false, error: "Endpoint not found" }, 404); } catch (error) { return json({ success: false, error: error.message || "Worker error" }, 500); } } }; // ====================================== // COLLECTIBLE SCANNER // ====================================== async function getCollectibles(userId) { let cursor = null; let totalRAP = 0; const items = []; const warnings = []; // Safety limit so a huge inventory // doesn't hammer the Roblox API. const MAX_PAGES = 20; for (let page = 0; page < MAX_PAGES; page++) { let apiUrl = `${ROBLOX_INVENTORY}/v1/users/${encodeURIComponent(userId)}` + `/assets/collectibles?assetType=All&sortOrder=Asc&limit=100`; if (cursor) { apiUrl += `&cursor=${encodeURIComponent(cursor)}`; } const response = await fetch(apiUrl); if (response.status === 403) { throw new Error( "This player's inventory is private or Roblox denied access." ); } if (!response.ok) { throw new Error( `Inventory API returned HTTP ${response.status}` ); } const data = await response.json(); for (const item of data.data || []) { const rap = Number( item.recentAveragePrice ); const validRAP = Number.isFinite(rap) && rap >= 0; if (validRAP) { totalRAP += rap; } else { warnings.push( `${item.name || "Unknown item"} has no RAP value.` ); } items.push({ assetId: item.assetId ?? null, userAssetId: item.userAssetId ?? null, name: item.name || "Unknown item", rap: validRAP ? rap : null, originalPrice: item.originalPrice ?? null, serialNumber: item.serialNumber ?? null, isOnHold: item.isOnHold ?? false }); } cursor = data.nextPageCursor; if (!cursor) { break; } } if (cursor) { warnings.push( "Inventory is larger than the scanner page limit, so the displayed RAP may be incomplete." ); } // Highest RAP first. items.sort((a, b) => { return (b.rap || 0) - (a.rap || 0); }); return { totalRAP, items, warnings: [...new Set(warnings)] }; } // ====================================== // RESPONSE HELPERS // ====================================== function corsHeaders() { return { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type" }; } function json(data, status = 200) { return new Response( JSON.stringify(data, null, 2), { status, headers: { "Content-Type": "application/json", ...corsHeaders() } } ); }
