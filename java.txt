const DISCORD_WEBHOOK =
    "PASTE_NEW_WEBHOOK_HERE";


const MIN_PLAYERS = 20;
const MAX_PAGES = 5;


const corsHeaders = {

    "Access-Control-Allow-Origin": "*",

    "Access-Control-Allow-Methods":
        "GET, OPTIONS",

    "Access-Control-Allow-Headers":
        "Content-Type",

    "Cache-Control":
        "no-store"
};


function json(data, status = 200){

    return new Response(
        JSON.stringify(data, null, 2),
        {
            status,

            headers:{
                "Content-Type":
                    "application/json; charset=utf-8",

                ...corsHeaders
            }
        }
    );

}


function scoreServer(server){

    const players =
        server.playing || 0;

    const maxPlayers =
        server.maxPlayers || 1;

    const occupancy =
        players / maxPlayers;

    let score =
        occupancy * 70;


    if(typeof server.ping === "number"){

        if(server.ping <= 60){
            score += 20;
        }

        else if(server.ping <= 100){
            score += 15;
        }

        else if(server.ping <= 150){
            score += 10;
        }

        else{
            score += 5;
        }

    }


    if(typeof server.fps === "number"){

        if(server.fps >= 55){
            score += 10;
        }

        else if(server.fps >= 40){
            score += 7;
        }

        else{
            score += 3;
        }

    }


    return Math.round(score * 100) / 100;

}


async function getServers(placeId){

    const servers =
        new Map();

    let cursor = "";


    for(
        let page = 0;
        page < MAX_PAGES;
        page++
    ){

        let api =
            `https://games.roblox.com/v1/games/${placeId}/servers/Public` +
            `?sortOrder=Desc&limit=100`;


        if(cursor){

            api +=
                `&cursor=${encodeURIComponent(cursor)}`;

        }


        const response =
            await fetch(api);


        if(!response.ok){

            throw new Error(
                `Roblox API returned ${response.status}`
            );

        }


        const result =
            await response.json();


        for(
            const server of result.data || []
        ){

            if(!server.id){
                continue;
            }


            if(
                (server.playing || 0)
                < MIN_PLAYERS
            ){
                continue;
            }


            servers.set(
                server.id,
                {

                    id:
                        server.id,

                    playing:
                        server.playing || 0,

                    maxPlayers:
                        server.maxPlayers || 0,

                    ping:
                        typeof server.ping === "number"
                        ? server.ping
                        : null,

                    fps:
                        typeof server.fps === "number"
                        ? server.fps
                        : null

                }
            );

        }


        cursor =
            result.nextPageCursor || "";


        if(!cursor){
            break;
        }

    }


    return [...servers.values()]

        .map(server => ({

            ...server,

            score:
                scoreServer(server)

        }))

        .sort((a,b) => {

            if(
                b.score !== a.score
            ){

                return b.score - a.score;

            }

            return b.playing - a.playing;

        });

}


function getJoinUrl(
    placeId,
    serverId
){

    return (
        "https://www.roblox.com/games/start" +
        `?placeId=${placeId}` +
        `&gameInstanceId=${encodeURIComponent(serverId)}`
    );

}


async function sendDiscord(
    game,
    server
){

    if(
        !DISCORD_WEBHOOK ||
        DISCORD_WEBHOOK ===
        "PASTE_NEW_WEBHOOK_HERE"
    ){

        return;

    }


    const joinUrl =
        getJoinUrl(
            game.placeId,
            server.id
        );


    const payload = {

        username:
            "DONIE Rich Finder",

        embeds:[{

            title:
                `${game.emoji} ${game.name} Server Detected`,

            description:
                "🟢 **Qualifying server detected by DONIE Rich Finder!**",

            color:
                5793266,

            fields:[

                {
                    name:"👥 Players",
                    value:
                        `${server.playing}/${server.maxPlayers}`,
                    inline:true
                },

                {
                    name:"📶 Ping",
                    value:
                        server.ping === null
                        ? "N/A"
                        : `${server.ping} ms`,
                    inline:true
                },

                {
                    name:"⚡ FPS",
                    value:
                        server.fps === null
                        ? "N/A"
                        : `${server.fps}`,
                    inline:true
                },

                {
                    name:"💎 Server Score",
                    value:
                        `${server.score}`,
                    inline:true
                },

                {
                    name:"🎮 Game",
                    value:
                        game.name,
                    inline:true
                },

                {
                    name:"🆔 Server ID",
                    value:
                        `\`${server.id}\``,
                    inline:false
                }

            ],

            url:
                joinUrl,

            footer:{
                text:
                    "DONIE Rich Finder"
            },

            timestamp:
                new Date().toISOString()

        }]

    };


    try{

        await fetch(
            DISCORD_WEBHOOK,
            {

                method:"POST",

                headers:{
                    "Content-Type":
                        "application/json"
                },

                body:
                    JSON.stringify(payload)

            }
        );

    }catch(error){

        console.error(
            "Discord webhook error:",
            error
        );

    }

}


export default {

    async fetch(request){

        if(
            request.method ===
            "OPTIONS"
        ){

            return new Response(
                null,
                {
                    status:204,
                    headers:
                        corsHeaders
                }
            );

        }


        const url =
            new URL(request.url);


        /*
        ================================
        HEALTH CHECK
        ================================
        */

        if(
            url.pathname === "/" ||
            url.pathname === "/health"
        ){

            return json({

                online:true,

                service:
                    "DONIE Rich Finder",

                games:
                    GAMES

            });

        }


        /*
        ================================
        SERVER API
        ================================
        */

        if(
            url.pathname === "/servers"
        ){

            const gameKey =
                url.searchParams.get("game")
                ||
                DEFAULT_GAME;


            const game =
                GAMES[gameKey];


            if(!game){

                return json(

                    {
                        success:false,

                        error:
                            "Unknown game. Use adoptme, bladeball, or mm2."

                    },

                    400

                );

            }


            try{

                const servers =
                    await getServers(
                        game.placeId
                    );


                /*
                Send the best server
                to Discord.
                */

                if(
                    servers.length > 0
                ){

                    await sendDiscord(
                        game,
                        servers[0]
                    );

                }


                return json({

                    success:true,

                    game:{

                        key:
                            gameKey,

                        name:
                            game.name,

                        emoji:
                            game.emoji,

                        placeId:
                            game.placeId

                    },

                    count:
                        servers.length,

                    data:
                        servers.map(
                            server => ({

                                ...server,

                                joinUrl:
                                    getJoinUrl(
                                        game.placeId,
                                        server.id
                                    )

                            })
                        )

                });


            }catch(error){

                return json(

                    {

                        success:false,

                        error:
                            error.message,

                        data:[]

                    },

                    502

                );

            }

        }


        return json(

            {

                success:false,

                error:
                    "Endpoint not found."

            },

            404

        );

    }

};
