async function searchPlayer() {
  const username = document.getElementById("username").value.trim();
  const result = document.getElementById("result");

  if (!username) {
    result.innerHTML = "<p>Enter a username.</p>";
    return;
  }

  result.innerHTML = "<p>Searching...</p>";

  try {
    const response = await fetch(
      "https://users.roblox.com/v1/usernames/users",
      {
        method: "POST",
        headers: {
          "Content-Type": "application/json"
        },
        body: JSON.stringify({
          usernames: [username],
          excludeBannedUsers: false
        })
      }
    );

    const data = await response.json();

    if (!data.data || data.data.length === 0) {
      result.innerHTML = "<p>Player not found.</p>";
      return;
    }

    const player = data.data[0];

    const avatarResponse = await fetch(
      `https://thumbnails.roblox.com/v1/users/avatar-headshot?userIds=${player.id}&size=150x150&format=Png&isCircular=false`
    );

    const avatarData = await avatarResponse.json();
    const avatar = avatarData.data[0].imageUrl;

    result.innerHTML = `
      <div class="card">
        <img class="avatar" src="${avatar}">

        <h2>${player.displayName}</h2>
        <p>@${player.name}</p>

        <div class="stat">
          <b>User ID:</b> ${player.id}
        </div>

        <div class="stat">
          <b>Status:</b> Public Roblox data only
        </div>

        <a
          class="join"
          href="https://www.roblox.com/users/${player.id}/profile"
          target="_blank">
          Open Roblox Profile
        </a>
      </div>
    `;

  } catch (error) {
    console.error(error);
    result.innerHTML = "<p>Something went wrong.</p>";
  }
}
