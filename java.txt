const ROBLOX_API = "https://games.roblox.com";
const USERS_API = "https://users.roblox.com";
const INVENTORY_API = "https://inventory.roblox.com";
const ECONOMY_API = "https://economy.roblox.com";

export default {
  async fetch(request) {
    const url = new URL(request.url);

    // CORS
    if (request.method === "OPTIONS") {
      return new Response(null, {
        headers: corsHeaders()
      });
    }

    try {
      // =========================
      // HEALTH CHECK
      // =========================
      if (url.pathname === "/") {
        return json({
          success: true,
          message: "DONIE Worker is online"
        });
      }

      // =========================
      // SERVER FINDER
      // =========================
      if (url.pathname === "/servers") {
        const placeId =
          url.searchParams.get("placeId") || "920587237";

        const limit =
          Math.min(
            Number(url.searchParams.get("limit") || 100),
            100
          );

        const apiUrl =
          `${ROBLOX_API}/v1/games/${encodeURIComponent(placeId)}/servers/Public` +
          `?sortOrder=Desc&limit=${limit}`;

        const response = await fetch(apiUrl, {
          headers: {
            "User-Agent": "DONIE-Rich-Finder/1.0"
          }
        });

        if (!response.ok) {
          throw new Error(
            `Roblox server API returned ${response.status}`
          );
        }

        const data = await response.json();

        const servers = (data.data || []).map(server => ({
          id: server.id,
          playing: Number(server.playing || 0),
          maxPlayers: Number(server.maxPlayers || 0),
          ping: server.ping ?? null,
          fps: server.fps ?? null
        }));

        return json({
          success: true,
          count: servers.length,
          data: servers
        });
      }

      // =========================
      // USERNAME -> USER ID
      // =========================
      if (url.pathname === "/user") {
        const username =
          url.searchParams.get("username");

        if (!username) {
          return json({
            success: false,
            error: "Missing username"
          }, 400);
        }

        const response = await fetch(
          `${USERS_API}/v1/usernames/users`,
          {
            method: "POST",
            headers: {
              "Content-Type": "application/json"
            },
            body: JSON.stringify({
              usernames: [username],
              excludeBannedUsers: false
            })
          }
        );

        if (!response.ok) {
          throw new Error(
            `Roblox user API returned ${response.status}`
          );
        }

        const data = await response.json();
        const user = data.data?.[0];

        if (!user) {
          return json({
            success: false,
            error: "Roblox user not found"
          }, 404);
        }

        return json({
          success: true,
          user: {
            id: user.id,
            name: user.name,
            displayName: user.displayName
          }
        });
      }

      // =========================
      // USER PROFILE
      // =========================
      if (url.pathname === "/profile") {
        const userId =
          url.searchParams.get("userId");

        if (!userId) {
          return json({
            success: false,
            error: "Missing userId"
          }, 400);
        }

        const response = await fetch(
          `${USERS_API}/v1/users/${encodeURIComponent(userId)}`
        );

        if (!response.ok) {
          return json({
            success: false,
            error: "User not found"
          }, 404);
        }

        const user = await response.json();

        return json({
          success: true,
          user: {
            id: user.id,
            name: user.name,
            displayName: user.displayName,
            description: user.description || "",
            created: user.created
          }
        });
      }

      // =========================
      // INVENTORY / COLLECTIBLES
      // =========================
      if (url.pathname === "/inventory") {
        const userId =
          url.searchParams.get("userId");

        if (!userId) {
          return json({
            success: false,
            error: "Missing userId"
          }, 400);
        }

        const assetTypes = [
          "Hat",
          "HairAccessory",
          "FaceAccessory",
          "NeckAccessory",
          "ShoulderAccessory",
          "FrontAccessory",
          "BackAccessory",
          "WaistAccessory",
          "Face",
          "Shirt",
          "Pants",
          "Gear"
        ];

        let items = [];

        // Try Roblox inventory endpoints.
        for (const assetType of assetTypes) {
          try {
            const endpoint =
              `${INVENTORY_API}/v1/users/${encodeURIComponent(userId)}` +
              `/items/${encodeURIComponent(assetType)}` +
              `?sortOrder=Desc&limit=100`;

            const response = await fetch(endpoint);

            if (!response.ok) {
              continue;
            }

            const data = await response.json();

            for (const item of data.data || []) {
              items.push({
                id: item.id,
                name: item.name,
                assetType: assetType,
                created: item.created || null
              });
            }
          } catch (e) {
            // Continue with the other asset types.
          }
        }

        // Remove duplicate IDs.
        const unique = [];
        const seen = new Set();

        for (const item of items) {
          if (!seen.has(item.id)) {
            seen.add(item.id);
            unique.push(item);
          }
        }

        return json({
          success: true,
          userId: Number(userId),
          count: unique.length,
          items: unique,
          note:
            "Only publicly available inventory information is returned. " +
            "Private or unavailable inventory cannot be bypassed."
        });
      }

      // =========================
      // AVATAR ITEMS
      // =========================
      if (url.pathname === "/avatar") {
        const userId =
          url.searchParams.get("userId");

        if (!userId) {
          return json({
            success: false,
            error: "Missing userId"
          }, 400);
        }

        const response = await fetch(
          `${USERS_API}/v1/users/${encodeURIComponent(userId)}/avatar`
        );

        if (!response.ok) {
          throw new Error(
            `Avatar API returned ${response.status}`
          );
        }

        const data = await response.json();

        return json({
          success: true,
          data
        });
      }

      return json({
        success: false,
        error: "Endpoint not found"
      }, 404);

    } catch (error) {
      return json({
        success: false,
        error: error.message || "Worker error"
      }, 500);
    }
  }
};

function corsHeaders() {
  return {
    "Access-Control-Allow-Origin": "*",
    "Access-Control-Allow-Methods": "GET, POST, OPTIONS",
    "Access-Control-Allow-Headers": "Content-Type"
  };
}

function json(data, status = 200) {
  return new Response(
    JSON.stringify(data, null, 2),
    {
      status,
      headers: {
        "Content-Type": "application/json",
        ...corsHeaders()
      }
    }
  );
}
